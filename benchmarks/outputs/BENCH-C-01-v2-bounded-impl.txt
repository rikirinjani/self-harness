```python
import json
from pathlib import Path
from collections import defaultdict
from typing import Dict, List, Any


def compute_aggregate_stats(results_dir: Path) -> Dict[str, Any]:
    """Compute aggregate statistics from benchmark result files.

    Args:
        results_dir: Path to directory containing BENCH-*.json result files.

    Returns:
        dict with total_benchmarks, avg_score, highest, lowest, pass_rate,
        and category_breakdown as specified.
    """
    results_dir = Path(results_dir)
    result_files = sorted(results_dir.glob("BENCH-*.json"))

    scores: List[Dict[str, Any]] = []
    category_data: Dict[str, List[Dict[str, Any]]] = defaultdict(list)
    total = 0
    passed = 0
    highest: Dict[str, Any] = {"id": "", "score": 0.0, "title": ""}
    lowest: Dict[str, Any] = {"id": "", "score": 0.0, "title": ""}

    for rf in result_files:
        try:
            with open(rf, encoding="utf-8") as f:
                data = json.load(f)
        except (json.JSONDecodeError, OSError):
            continue

        bid = data.get("benchmark_id", "unknown")
        title = data.get("task", "")
        avg = data.get("scores", {}).get("average")
        outcome = data.get("execution", {}).get("outcome", "")
        duration = data.get("execution", {}).get("duration_s", 0)

        if avg is None:
            continue

        cat_letter = bid.split("-")[1].lower() if "-" in bid else "?"
        cat_map = {"r": "research", "c": "coding", "p": "planning", "o": "operational"}
        category = cat_map.get(cat_letter, "unknown")

        entry = {"id": bid, "score": avg, "title": title, "duration": duration}
        scores.append(entry)
        category_data[category].append(entry)
        total += 1

        if outcome == "pass":
            passed += 1

        if highest["id"] == "" or avg > highest["score"]:
            highest = {"id": bid, "score": avg, "title": title}
        if lowest["id"] == "" or avg < lowest["score"]:
            lowest = {"id": bid, "score": avg, "title": title}

    if not scores:
        return {
            "total_benchmarks": 0,
            "avg_score": 0.0,
            "highest": {"id": "", "score": 0.0, "title": ""},
            "lowest": {"id": "", "score": 0.0, "title": ""},
            "pass_rate": 0.0,
            "category_breakdown": [
                {"category": c, "count": 0, "avg_score": 0.0, "avg_duration": 0.0}
                for c in ["research", "coding", "planning", "operational"]
            ],
        }

    avg_score = round(sum(s["score"] for s in scores) / len(scores), 2)
    pass_rate = round(passed / total * 100, 1) if total > 0 else 0.0

    category_breakdown = []
    for cat in ["research", "coding", "planning", "operational"]:
        entries = category_data.get(cat, [])
        if entries:
            cat_avg = round(sum(e["score"] for e in entries) / len(entries), 2)
            cat_dur = sum(e["duration"] for e in entries)
            category_breakdown.append({
                "category": cat,
                "count": len(entries),
                "avg_score": cat_avg,
                "avg_duration": round(cat_dur / len(entries), 1),
            })
        else:
            category_breakdown.append({
                "category": cat, "count": 0, "avg_score": 0.0, "avg_duration": 0.0,
            })

    return {
        "total_benchmarks": total,
        "avg_score": avg_score,
        "highest": highest,
        "lowest": lowest,
        "pass_rate": pass_rate,
        "category_breakdown": category_breakdown,
    }


if __name__ == "__main__":
    test_dir = Path(__file__).resolve().parent / "results"
    stats = compute_aggregate_stats(test_dir)
    assert isinstance(stats["total_benchmarks"], int)
    assert isinstance(stats["avg_score"], float)
    assert stats["highest"]["score"] >= stats["lowest"]["score"]
    assert 0 <= stats["pass_rate"] <= 100
    assert len(stats["category_breakdown"]) == 4
    assert "title" in stats["highest"] and "title" in stats["lowest"]
    print("[PASS] All assertions passed")
```
